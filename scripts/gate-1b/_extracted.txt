/**
 * The established numeric tolerance, used ONLY where re-association is
 * legitimate: sums, differences of sums, and derived ratios. Never on a
 * captured scalar compared to itself — that stays at zero drift.
 */
const NOISE = 1e-9;
const close = (x: number, y: number) =>
  x === y || Math.abs(x - y) <= NOISE * Math.max(1, Math.abs(x), Math.abs(y));

/**
 * Per-tier per-SKU scalars a freight contribution reaches, and may therefore
 * follow the shipment to another product.
 *
 * DELIBERATELY ENUMERATED, not pattern-matched. A regex over field names would
 * silently enrol the next field somebody adds, and enrolment is exemption from
 * the strict half — the one direction in which a mistake here is dangerous.
 */
const ATTRIBUTION_SUMMABLE = [
  "totalLandedFreightBeforeMarkup",
  "totalLandedFreightWithMarkup",
  "totalContainerFreightBeforeMarkup",
  "totalDutyTariffBeforeMarkup",
  "freightContainerMarkupSumPerUnit",
  "freightDutyTariffMarkupSumPerUnit",
  "contributionCostPerUnit",
  "sellBeforeAdjustmentPerUnit",
  "adjDeltaPerUnit",
  "sellAfterAdjustmentPerUnit",
  "liftDeltaPerUnit",
  "sellAfterLiftPerUnit",
  "overrideDeltaPerUnit",
  "computedSellPerUnit",
  "requiredSellPerUnit",
  "revenue",
  "cost",
] as const;

/** Ratios and verdicts. Not summable, so checked for internal consistency. */
const ATTRIBUTION_DERIVED = ["marginPct", "marginStatus", "competitiveStatus"] as const;

const ATTRIBUTION_FIELDS = new Set<string>([
  ...ATTRIBUTION_SUMMABLE,
  ...ATTRIBUTION_DERIVED,
]);

type Payload = {
  tiers?: { tierId: string; qty: number }[];
  skuRollups?: {
    skuId: string;
    skuRole: string;
    parentSkuId: string | null;
    perTier?: Record<string, unknown>[];
  }[];
};

/**
 * Aggregates are summed OVER the rows attribution may redistribute, so moving a
 * shipment between products changes the order the tier total accumulates in.
 * IEEE-754 addition is not associative, and the measured consequence is a
 * last-bit difference: 5.8e-11 on a $478,295 total, or 1.2e-16 relative.
 *
 * Zero drift is therefore not achievable on these while attribution is
 * permitted to move — not because any arithmetic changed, but because the
 * addends did. They are quantized to 12 SIGNIFICANT DIGITS on both sides, which
 * is ~1e-12 relative: tighter than the 1e-9 the rest of the gate calls its
 * established tolerance, ~10,000x tighter than a hundredth of a cent on the
 * largest total in the basket, and ~10,000x looser than the noise it absorbs.
 * The 5,880 movement this repair fixed clears it by thirteen orders of
 * magnitude.
 *
 * SCOPE IS THE POINT: only `quoteRollup` and `quoteSummary`. Every per-SKU
 * scalar outside the attribution set, and all identity and membership, stays at
 * full 17-digit zero drift.
 */
const AGGREGATE_SCOPES = new Set(["quoteRollup", "quoteSummary"]);
const SIG_DIGITS = 12;

function quantizeAggregates(node: unknown, scope: string | null): unknown {
  if (typeof node === "number") {
    return scope !== null && Number.isFinite(node)
      ? Number(node.toPrecision(SIG_DIGITS))
      : node;
  }
  if (Array.isArray(node)) return node.map((v) => quantizeAggregates(v, scope));
  if (node !== null && typeof node === "object") {
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(node as Record<string, unknown>))
      out[k] = quantizeAggregates(v, scope ?? (AGGREGATE_SCOPES.has(k) ? k : null));
    return out;
  }
  return node;
}

/** The payload with the attribution-bearing per-SKU scalars removed. */
function strictHalf(payload: unknown): unknown {
  const p = JSON.parse(JSON.stringify(payload ?? null)) as Payload | null;
  if (!p || typeof p !== "object") return p;
  for (const s of p.skuRollups ?? [])
    for (const pt of s.perTier ?? [])
      for (const f of ATTRIBUTION_FIELDS) delete pt[f];
  return quantizeAggregates(p, null);
}

type Cell = {
  skuId: string;
  skuRole: string;
  topLevel: boolean;
  tierId: string;
  v: Record<string, number | string | null>;
};

function cells(payload: unknown): Map<string, Cell> {
  const p = (payload ?? {}) as Payload;
  const out = new Map<string, Cell>();
  for (const s of p.skuRollups ?? [])
    for (const pt of s.perTier ?? []) {
      const tierId = String(pt.tierId);
      out.set(`${s.skuId}::${tierId}`, {
        skuId: s.skuId,
        skuRole: s.skuRole,
        topLevel: s.parentSkuId === null,
        tierId,
        v: pt as Record<string, number | string | null>,
      });
    }
  return out;
}

const numOf = (v: unknown) => (typeof v === "number" ? v : 0);

/**
 * Is every per-SKU movement on this quote a REATTRIBUTION of freight?
 *
 * Returns the violations. Empty means the products exchanged a shipment and did
 * nothing else — which Pattern 58 permits, and which the aggregates then have to
 * confirm.
 */
function attributionViolations(base: unknown, cur: unknown): string[] {
  const out: string[] = [];
  const B = cells(base);
  const C = cells(cur);
  const tierQty = new Map<string, number>();
  for (const t of ((cur ?? {}) as Payload).tiers ?? []) tierQty.set(t.tierId, t.qty);

  // Membership is the strict half's job, but a key present on one side only
  // would make every check below vacuous, so it is refused here too.
  for (const k of new Set([...B.keys(), ...C.keys()]))
    if (!B.has(k) || !C.has(k)) out.push(`cell ${k} present on one side only`);
  if (out.length > 0) return out;

  // ── 1 · the same freight, possibly held elsewhere ────────────────────────
  const tiers = new Set([...C.values()].map((c) => c.tierId));
  for (const tierId of tiers)
    for (const f of ["totalLandedFreightBeforeMarkup", "totalLandedFreightWithMarkup"] as const) {
      const ms = (m: Map<string, Cell>) =>
        [...m.values()]
          .filter((c) => c.tierId === tierId && c.skuRole === "leaf")
          .map((c) => numOf(c.v[f]))
          .sort((a, b) => a - b);
      const x = ms(B);
      const y = ms(C);
      if (x.length !== y.length || !x.every((v, i) => close(v, y[i])))
        out.push(`tier ${tierId}: the ${f} multiset changed — freight itself moved, not its owner`);
    }

  // ── 2 · each row moved by exactly its own freight, and by nothing else ───
  for (const [k, c] of C) {
    const b = B.get(k)!;
    const dCost = numOf(c.v.totalLandedFreightBeforeMarkup) - numOf(b.v.totalLandedFreightBeforeMarkup);
    const dSell = numOf(c.v.totalLandedFreightWithMarkup) - numOf(b.v.totalLandedFreightWithMarkup);
    const q = tierQty.get(c.tierId) ?? 0;
    const anchored: Array<[string, number, number]> = [
      ["contributionCostPerUnit", numOf(c.v.contributionCostPerUnit) - numOf(b.v.contributionCostPerUnit), dCost],
      ["sellBeforeAdjustmentPerUnit", numOf(c.v.sellBeforeAdjustmentPerUnit) - numOf(b.v.sellBeforeAdjustmentPerUnit), dSell],
      ["cost", numOf(c.v.cost) - numOf(b.v.cost), dCost * q],
      ["revenue", numOf(c.v.revenue) - numOf(b.v.revenue), (numOf(c.v.requiredSellPerUnit) - numOf(b.v.requiredSellPerUnit)) * q],
    ];
    for (const [f, actual, permitted] of anchored)
      if (!close(actual, permitted))
        out.push(
          `${k.slice(0, 8)}… ${f} moved ${actual} but its freight moved ${permitted} — ` +
            `a movement this row's attribution does not account for`,
        );
  }

  // ── 3 · the sums are conserved, over leaves and over top-level rows ──────
  for (const tierId of tiers)
    for (const [scope, pick] of [
      ["leaves", (c: Cell) => c.skuRole === "leaf"],
      ["top-level", (c: Cell) => c.topLevel],
    ] as const)
      for (const f of ATTRIBUTION_SUMMABLE) {
        const sum = (m: Map<string, Cell>) =>
          [...m.values()]
            .filter((c) => c.tierId === tierId && pick(c))
            .reduce((a, c) => a + numOf(c.v[f]), 0);
        const x = sum(B);
        const y = sum(C);
        if (!close(x, y))
          out.push(`tier ${tierId} ${scope}: ${f} sum ${x} -> ${y} — redistribution did not conserve`);
      }

  // ── 4 · each row's margin still follows from its own sell and cost ───────
  for (const [k, c] of C) {
    const sell = numOf(c.v.requiredSellPerUnit);
    const cost = numOf(c.v.contributionCostPerUnit);
    const m = c.v.marginPct;
    if (sell > 0 && typeof m === "number" && !close(m, (sell - cost) / sell))
      out.push(`${k.slice(0, 8)}… marginPct ${m} does not follow from its own sell ${sell} and cost ${cost}`);
  }

  return out;
}
